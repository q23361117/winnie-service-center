    async function registerMember(event){
      event.preventDefault();
      clearMemberError();

      const submitButton = document.getElementById(memberSubmitButton);
      const originalText = submitButton.textContent;

      const name = cleanText(document.getElementById(memberName).value);
      const phone = normalizePhone(document.getElementById(memberPhone).value);
      const consent = document.getElementById(memberConsent).checked;

      if(!name){
        showMemberError(請輸入姓名或稱呼。);
        return;
      }

      if(!^09d{8}$.test(phone)){
        showMemberError(手機號碼格式不正確，請輸入 09 開頭的 10 碼號碼。);
        return;
      }

      if(!consent){
        showMemberError(請先勾選同意會員資料使用說明。);
        return;
      }

      submitButton.disabled = true;
      submitButton.textContent = 建立會員中…;

      try{
        const profile = await ensureLiffProfile();

        if(!profile  !profile.userId){
          throw new Error(請從維尼車隊 LINE 官方的服務選單開啟此頁。);
        }

        const phoneIndexRef = db.collection(memberPhoneIndex).doc(phone);
        const lineIndexRef = db.collection(memberLineIndex).doc(profile.userId);
        const counterRef = db.collection(systemCounters).doc(members);

        const result = await db.runTransaction(async transaction = {
          const [phoneIndexSnapshot, lineIndexSnapshot, counterSnapshot] =
            await Promise.all([
              transaction.get(phoneIndexRef),
              transaction.get(lineIndexRef),
              transaction.get(counterRef)
            ]);

          if(lineIndexSnapshot.exists){
            return {
              alreadyExists true,
              memberCode lineIndexSnapshot.data().memberCode
            };
          }

          if(phoneIndexSnapshot.exists){
            throw new Error(這支手機號碼已經加入會員，請聯絡客服協助確認。);
          }

          const nextNumber = counterSnapshot.exists
             Number(counterSnapshot.data().nextNumber  1)
             1;

          const memberCode = formatMemberCode(nextNumber);
          const memberRef = db.collection(members).doc(memberCode);

          transaction.set(memberRef, {
            memberCode,
            name,
            phone,
            lineUserId profile.userId,
            lineDisplayName profile.displayName  ,
            linePictureUrl profile.pictureUrl  ,
            points 0,
            rideCredit 0,
            totalRides 0,
            status active,
            note ,
            source service_center_liff,
            createdAt firebase.firestore.FieldValue.serverTimestamp(),
            updatedAt firebase.firestore.FieldValue.serverTimestamp()
          });

          transaction.set(phoneIndexRef, {
            memberCode,
            createdAt firebase.firestore.FieldValue.serverTimestamp()
          });

          transaction.set(lineIndexRef, {
            memberCode,
            createdAt firebase.firestore.FieldValue.serverTimestamp()
          });

          transaction.set(counterRef, {
            nextNumber nextNumber + 1,
            updatedAt firebase.firestore.FieldValue.serverTimestamp()
          }, {
            merge true
          });

          return {
            alreadyExists false,
            memberCode
          };
        });

        const memberSnapshot = await db.collection(members).doc(result.memberCode).get();

        if(!memberSnapshot.exists){
          throw new Error(會員建立完成，但暫時無法讀取會員資料。);
        }

        currentMember = {
          id memberSnapshot.id,
          ...memberSnapshot.data()
        };

        renderMemberCard(currentMember);
        setMemberPanelState(existing);

`🎉【維尼車隊｜會員申請完成】

👤 會員姓名：${currentMember.name}
🎫 會員編號：${currentMember.memberCode}

⭐ 搭乘完成可累積 1 點

📲 後續叫車時，請將會員編號提供給派車小姐：
${currentMember.memberCode}`;

        renderOutput(member, successMessage);

        if(window.liff && liffReady && liff.isInClient()){
          try{

    async function copyText(btn){
      const text = btn.closest(.output).querySelector(textarea).value;

      if(navigator.clipboard && window.isSecureContext){
        try{
          await navigator.clipboard.writeText(text);
          copied(btn);
        }catch(e){
          fallbackCopy(btn,text);
        }
      }else{
        fallbackCopy(btn,text);
      }
    }

    function fallbackCopy(btn,text){
      const ta = btn.closest(.output).querySelector(textarea);
      ta.focus();
      ta.select();
      document.execCommand(copy);
      copied(btn);
    }
